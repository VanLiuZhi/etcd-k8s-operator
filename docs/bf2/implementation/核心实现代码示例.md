# 核心实现代码示例

## 1. 调谐协调器实现

```go
package coordinator

import (
	"context"
	"fmt"
	"sync"
	"time"

	"github.com/go-logr/logr"
	"k8s.io/apimachinery/pkg/types"
	"k8s.io/client-go/util/retry"
)

// 调谐协调器 - 核心并发控制组件
type ReconcilerCoordinator struct {
	// 配置
	config *Config

	// 日志记录器
	logger logr.Logger

	// 调谐锁 - 确保同一时间只有一个调谐操作
	reconcileMutex sync.Mutex

	// 当前调谐状态
	currentState *ReconcileState

	// 等待队列 - 存储待处理的调谐事件
	pendingQueue []*ReconcileEvent

	// 事件通知通道
	notifyCh chan struct{}

	// 状态存储
	stateStore StateStore

	// 重试管理器
	retryManager *RetryManager
}

// 调谐状态
type ReconcileState struct {
	// 是否正在调谐中
	IsReconciling bool `json:"is_reconciling"`

	// 当前调谐类型
	CurrentType ReconcileType `json:"current_type"`

	// 当前调谐的事件ID
	CurrentEventID string `json:"current_event_id"`

	// 开始时间
	StartTime time.Time `json:"start_time"`

	// 状态版本
	Version int64 `json:"version"`

	// 调谐进度
	Progress *ReconcileProgress `json:"progress,omitempty"`
}

// 调谐事件
type ReconcileEvent struct {
	// 事件ID
	EventID string `json:"event_id"`

	// 事件类型
	Type ReconcileType `json:"type"`

	// 优先级
	Priority Priority `json:"priority"`

	// 关联的集群
	ClusterKey types.NamespacedName `json:"cluster_key"`

	// 事件数据
	Data interface{} `json:"data"`

	// 重试配置
	MaxRetries int `json:"max_retries"`
	RetryCount int `json:"retry_count"`

	// 时间戳
	CreatedAt time.Time `json:"created_at"`
	ExpiresAt time.Time `json:"expires_at,omitempty"`

	// 回调函数
	Callback func(*ReconcileEvent) error `json:"-"`
}

// 调谐类型
type ReconcileType string

const (
	ReconcileTypeExternal ReconcileType = "external"  // 外部CR变化触发
	ReconcileTypeInternal ReconcileType = "internal"  // 内部健康检查触发
	ReconcileTypeRecovery ReconcileType = "recovery"  // 故障恢复触发
)

// 优先级定义
type Priority int

const (
	PriorityCritical Priority = iota + 1  // 关键：故障恢复
	PriorityHigh                         // 高：扩缩容
	PriorityMedium                       // 中：配置更新
	PriorityLow                          // 低：状态检查
)

// 新建调谐协调器
func NewReconcilerCoordinator(config *Config, logger logr.Logger) *ReconcilerCoordinator {
	return &ReconcilerCoordinator{
		config:        config,
		logger:        logger,
		currentState:  &ReconcileState{},
		notifyCh:      make(chan struct{}, 1),
		stateStore:    NewCRStateStore(config.Client, config.Cluster),
		retryManager:  NewRetryManager(config.MaxRetries, config.BaseRetryDelay),
	}
}

// 处理调谐请求 - 主要入口点
func (rc *ReconcilerCoordinator) Reconcile(ctx context.Context, req *ReconcileRequest) error {
	rc.logger.Info("开始处理调谐请求",
		"event_type", req.Type,
		"priority", req.Priority,
		"cluster", req.ClusterKey)

	// 创建调谐事件
	event := &ReconcileEvent{
		EventID:     generateEventID(),
		Type:        req.Type,
		Priority:    req.Priority,
		ClusterKey:  req.ClusterKey,
		Data:        req.Data,
		MaxRetries:   rc.config.MaxRetries,
		CreatedAt:   time.Now(),
		ExpiresAt:   time.Now().Add(rc.config.EventTimeout),
		Callback:    req.Callback,
	}

	// 尝试获取调谐锁
	if rc.tryAcquireReconcileLock(event) {
		// 成功获取锁，开始处理
		return rc.processReconcileEvent(ctx, event)
	}

	// 获取锁失败，加入等待队列
	return rc.addToPendingQueue(event)
}

// 尝试获取调谐锁
func (rc *ReconcilerCoordinator) tryAcquireReconcileLock(event *ReconcileEvent) bool {
	rc.reconcileMutex.Lock()
	defer rc.reconcileMutex.Unlock()

	// 如果当前没有调谐在进行，或者新事件优先级更高
	if !rc.currentState.IsReconciling || event.Priority > rc.getCurrentPriority() {
		rc.currentState = &ReconcileState{
			IsReconciling:   true,
			CurrentType:     event.Type,
			CurrentEventID:  event.EventID,
			StartTime:       time.Now(),
			Version:         rc.currentState.Version + 1,
			Progress:        &ReconcileProgress{Stage: "starting"},
		}
		return true
	}

	return false
}

// 处理调谐事件
func (rc *ReconcilerCoordinator) processReconcileEvent(ctx context.Context, event *ReconcileEvent) error {
	rc.logger.Info("开始处理调谐事件",
		"event_id", event.EventID,
		"type", event.Type,
		"priority", event.Priority)

	// 更新状态为处理中
	rc.updateReconcileProgress("processing", 0)

	// 执行调谐逻辑
	var err error
	switch event.Type {
	case ReconcileTypeExternal:
		err = rc.processExternalReconcile(ctx, event)
	case ReconcileTypeInternal:
		err = rc.processInternalReconcile(ctx, event)
	case ReconcileTypeRecovery:
		err = rc.processRecoveryReconcile(ctx, event)
	default:
		err = fmt.Errorf("unknown reconcile type: %s", event.Type)
	}

	// 处理结果
	if err != nil {
		rc.logger.Error(err, "调谐事件处理失败",
			"event_id", event.EventID,
			"retry_count", event.RetryCount)

		// 重试逻辑
		if event.RetryCount < event.MaxRetries {
			return rc.retryEvent(event, err)
		}

		// 达到最大重试次数，标记为失败
		rc.updateReconcileProgress("failed", 100)
		return fmt.Errorf("调谐事件处理失败，已达到最大重试次数: %w", err)
	}

	// 处理成功
	rc.updateReconcileProgress("completed", 100)
	rc.logger.Info("调谐事件处理成功",
		"event_id", event.EventID,
		"type", event.Type)

	// 释放锁并处理等待队列
	return rc.completeReconcileEvent(event)
}

// 处理外部调谐（CR变化）
func (rc *ReconcilerCoordinator) processExternalReconcile(ctx context.Context, event *ReconcileEvent) error {
	rc.logger.Info("处理外部调谐事件", "event_id", event.EventID)

	// 获取事件数据
	data, ok := event.Data.(*ExternalReconcileData)
	if !ok {
		return fmt.Errorf("invalid external reconcile data")
	}

	// 更新进度
	rc.updateReconcileProgress("updating_cluster_spec", 20)

	// 更新集群规格
	if err := rc.updateClusterSpec(data); err != nil {
		return fmt.Errorf("failed to update cluster spec: %w", err)
	}

	// 执行扩缩容操作
	rc.updateReconcileProgress("scaling_cluster", 50)
	if err := rc.scaleCluster(data); err != nil {
		return fmt.Errorf("failed to scale cluster: %w", err)
	}

	// 验证结果
	rc.updateReconcileProgress("verifying_result", 80)
	if err := rc.verifyClusterState(data); err != nil {
		return fmt.Errorf("failed to verify cluster state: %w", err)
	}

	return nil
}

// 处理内部调谐（健康检查）
func (rc *ReconcilerCoordinator) processInternalReconcile(ctx context.Context, event *ReconcileEvent) error {
	rc.logger.Info("处理内部调谐事件", "event_id", event.EventID)

	// 获取集群健康状态
	rc.updateReconcileProgress("checking_health", 20)
	healthStatus, err := rc.checkClusterHealth()
	if err != nil {
		return fmt.Errorf("failed to check cluster health: %w", err)
	}

	// 如果有故障节点，创建故障恢复事件
	if len(healthStatus.UnhealthyMembers) > 0 {
		rc.logger.Info("发现故障节点，创建故障恢复事件",
			"unhealthy_members", healthStatus.UnhealthyMembers)

		recoveryEvent := &ReconcileEvent{
			EventID:    generateEventID(),
			Type:       ReconcileTypeRecovery,
			Priority:   PriorityCritical,
			ClusterKey: event.ClusterKey,
			Data: &RecoveryData{
				UnhealthyMembers: healthStatus.UnhealthyMembers,
				HealthStatus:     healthStatus,
			},
			MaxRetries: rc.config.MaxRetries,
			CreatedAt:  time.Now(),
		}

		// 立即处理故障恢复事件
		return rc.processReconcileEvent(ctx, recoveryEvent)
	}

	// 更新状态
	rc.updateReconcileProgress("updating_status", 60)
	if err := rc.updateClusterStatus(healthStatus); err != nil {
		return fmt.Errorf("failed to update cluster status: %w", err)
	}

	return nil
}

// 处理故障恢复
func (rc *ReconcilerCoordinator) processRecoveryReconcile(ctx context.Context, event *ReconcileEvent) error {
	rc.logger.Info("处理故障恢复事件", "event_id", event.EventID)

	// 获取故障数据
	data, ok := event.Data.(*RecoveryData)
	if !ok {
		return fmt.Errorf("invalid recovery data")
	}

	// 对每个故障节点执行恢复
	for _, member := range data.UnhealthyMembers {
		rc.logger.Info("恢复故障节点", "member", member.Name)

		rc.updateReconcileProgress(fmt.Sprintf("recovering_member_%s", member.Name), 30)

		// 隔离故障节点
		if err := rc.isolateMember(member); err != nil {
			rc.logger.Error(err, "隔离节点失败", "member", member.Name)
			continue
		}

		// 创建新节点替换故障节点
		if err := rc.replaceMember(member); err != nil {
			rc.logger.Error(err, "替换节点失败", "member", member.Name)
			continue
		}

		// 验证新节点状态
		if err := rc.verifyNewMember(member); err != nil {
			rc.logger.Error(err, "验证新节点失败", "member", member.Name)
			continue
		}
	}

	// 验证集群整体健康状态
	rc.updateReconcileProgress("verifying_cluster_health", 90)
	if err := rc.verifyClusterHealth(); err != nil {
		return fmt.Errorf("failed to verify cluster health: %w", err)
	}

	return nil
}

// 完成调谐事件处理
func (rc *ReconcilerCoordinator) completeReconcileEvent(event *ReconcileEvent) error {
	rc.reconcileMutex.Lock()
	defer rc.reconcileMutex.Unlock()

	// 重置当前状态
	rc.currentState = &ReconcileState{
		Version: rc.currentState.Version + 1,
	}

	// 处理等待队列中的事件
	if len(rc.pendingQueue) > 0 {
		// 按优先级排序
		rc.sortPendingQueueByPriority()

		// 获取下一个事件
		nextEvent := rc.pendingQueue[0]
		rc.pendingQueue = rc.pendingQueue[1:]

		// 异步处理下一个事件
		go func() {
			if err := rc.processReconcileEvent(context.Background(), nextEvent); err != nil {
				rc.logger.Error(err, "处理等待队列事件失败",
					"event_id", nextEvent.EventID)
			}
		}()
	}

	return nil
}

// 重试事件
func (rc *ReconcilerCoordinator) retryEvent(event *ReconcileEvent, originalErr error) error {
	// 增加重试计数
	event.RetryCount++

	// 计算重试延迟
	delay := rc.retryManager.NextDelay(event.RetryCount)
	if delay == 0 {
		return fmt.Errorf("达到最大重试次数: %w", originalErr)
	}

	rc.logger.Info("重试调谐事件",
		"event_id", event.EventID,
		"retry_count", event.RetryCount,
		"delay", delay)

	// 延迟重试
	time.Sleep(delay)

	// 重新处理事件
	return rc.processReconcileEvent(context.Background(), event)
}

// 更新调谐进度
func (rc *ReconcilerCoordinator) updateReconcileProgress(stage string, percentage int) {
	rc.reconcileMutex.Lock()
	defer rc.reconcileMutex.Unlock()

	if rc.currentState.Progress == nil {
		rc.currentState.Progress = &ReconcileProgress{}
	}

	rc.currentState.Progress.Stage = stage
	rc.currentState.Progress.Percentage = percentage
	rc.currentState.Progress.LastUpdated = time.Now()

	// 持久化状态
	if err := rc.stateStore.SaveState(rc.currentState); err != nil {
		rc.logger.Error(err, "保存状态失败")
	}
}

// 获取当前优先级
func (rc *ReconcilerCoordinator) getCurrentPriority() Priority {
	switch rc.currentState.CurrentType {
	case ReconcileTypeRecovery:
		return PriorityCritical
	case ReconcileTypeExternal:
		return PriorityHigh
	case ReconcileTypeInternal:
		return PriorityLow
	default:
		return PriorityLow
	}
}

// 生成事件ID
func generateEventID() string {
	return fmt.Sprintf("evt-%d", time.Now().UnixNano())
}
```

## 2. 事件队列实现

```go
package queue

import (
	"container/heap"
	"context"
	"fmt"
	"sync"
	"time"

	"github.com/go-logr/logr"
)

// 优先级事件队列
type EventQueue struct {
	// 配置
	config *QueueConfig

	// 日志记录器
	logger logr.Logger

	// 优先级队列（使用堆实现）
	queue *EventHeap

	// 队列锁
	queueMutex sync.RWMutex

	// 当前处理的事件
	currentEvent *QueuedEvent

	// 事件历史
	history map[string]*EventRecord

	// 统计信息
	stats *QueueStats

	// 停止通道
	stopCh chan struct{}
}

// 队列配置
type QueueConfig struct {
	MaxQueueSize    int           `json:"max_queue_size"`
	BatchSize       int           `json:"batch_size"`
	BatchTimeout    time.Duration `json:"batch_timeout"`
	EventTimeout    time.Duration `json:"event_timeout"`
	RetryDelay      time.Duration `json:"retry_delay"`
	MaxRetries      int           `json:"max_retries"`
}

// 队列中的事件
type QueuedEvent struct {
	// 事件内容
	Event *ReconcileEvent

	// 入队时间
	EnqueuedAt time.Time

	// 优先级
	Priority int

	// 索引（用于堆操作）
	index int
}

// 事件堆实现
type EventHeap []*QueuedEvent

func (h EventHeap) Len() int { return len(h) }

func (h EventHeap) Less(i, j int) bool {
	// 优先级高的排在前面
	if h[i].Priority != h[j].Priority {
		return h[i].Priority > h[j].Priority
	}
	// 相同优先级按时间排序
	return h[i].EnqueuedAt.Before(h[j].EnqueuedAt)
}

func (h EventHeap) Swap(i, j int) {
	h[i], h[j] = h[j], h[i]
	h[i].index = i
	h[j].index = j
}

func (h *EventHeap) Push(x interface{}) {
	item := x.(*QueuedEvent)
	item.index = len(*h)
	*h = append(*h, item)
}

func (h *EventHeap) Pop() interface{} {
	old := *h
	n := len(old)
	item := old[n-1]
	old[n-1] = nil  // 避免内存泄漏
	item.index = -1 // 安全标记
	*h = old[0 : n-1]
	return item
}

// 新建事件队列
func NewEventQueue(config *QueueConfig, logger logr.Logger) *EventQueue {
	eq := &EventQueue{
		config:  config,
		logger:  logger,
		queue:   &EventHeap{},
		history: make(map[string]*EventRecord),
		stats:   &QueueStats{},
		stopCh:  make(chan struct{}),
	}

	// 初始化堆
	heap.Init(eq.queue)

	return eq
}

// 启动事件处理器
func (eq *EventQueue) Start(ctx context.Context) {
	eq.logger.Info("启动事件队列处理器")

	// 启动批处理器
	go eq.batchProcessor(ctx)

	// 启动清理器
	go eq.cleanupExpiredEvents(ctx)

	// 启动统计收集器
	go eq.statsCollector(ctx)
}

// 添加事件到队列
func (eq *EventQueue) Enqueue(event *ReconcileEvent) error {
	eq.queueMutex.Lock()
	defer eq.queueMutex.Unlock()

	// 检查队列是否已满
	if eq.queue.Len() >= eq.config.MaxQueueSize {
		return fmt.Errorf("事件队列已满")
	}

	// 创建队列事件
	qe := &QueuedEvent{
		Event:      event,
		EnqueuedAt: time.Now(),
		Priority:   eq.getPriorityValue(event.Priority),
	}

	// 加入堆
	heap.Push(eq.queue, qe)

	// 更新统计
	eq.stats.EnqueuedTotal++
	eq.stats.QueueSize = eq.queue.Len()

	eq.logger.Info("事件加入队列",
		"event_id", event.EventID,
		"type", event.Type,
		"priority", event.Priority,
		"queue_size", eq.queue.Len())

	return nil
}

// 批处理器
func (eq *EventQueue) batchProcessor(ctx context.Context) {
	batchTimer := time.NewTimer(eq.config.BatchTimeout)
	defer batchTimer.Stop()

	batch := make([]*QueuedEvent, 0, eq.config.BatchSize)

	for {
		select {
		case <-ctx.Done():
			return
		case <-eq.stopCh:
			return
		case <-batchTimer.C:
			// 处理当前批次
			if len(batch) > 0 {
				eq.processBatch(batch)
				batch = batch[:0]
			}
			batchTimer.Reset(eq.config.BatchTimeout)
		default:
			// 检查是否有新事件
			if eq.queue.Len() > 0 {
				eq.queueMutex.Lock()
				if eq.queue.Len() > 0 {
					// 获取事件
					event := heap.Pop(eq.queue).(*QueuedEvent)
					batch = append(batch, event)
					eq.stats.QueueSize = eq.queue.Len()

					// 如果批次满了，立即处理
					if len(batch) >= eq.config.BatchSize {
						batchTimer.Stop()
						batchTimer.Reset(0)
					}
				}
				eq.queueMutex.Unlock()
			}
			time.Sleep(100 * time.Millisecond) // 避免忙等待
		}
	}
}

// 处理批次事件
func (eq *EventQueue) processBatch(batch []*QueuedEvent) {
	eq.logger.Info("处理事件批次", "batch_size", len(batch))

	for _, event := range batch {
		eq.processSingleEvent(event)
	}
}

// 处理单个事件
func (eq *EventQueue) processSingleEvent(event *QueuedEvent) {
	startTime := time.Now()

	// 记录开始处理
	record := &EventRecord{
		EventID:    event.Event.EventID,
		EventType:  event.Event.Type,
		StartTime:  startTime,
		Priority:   event.Event.Priority,
		RetryCount: event.Event.RetryCount,
	}

	eq.history[event.Event.EventID] = record

	// 执行事件回调
	var err error
	if event.Event.Callback != nil {
		err = event.Event.Callback(event.Event)
	}

	// 记录处理结果
	record.EndTime = time.Now()
	record.Duration = record.EndTime.Sub(record.StartTime)
	record.Success = err == nil
	if err != nil {
		record.Error = err.Error()
	}

	// 更新统计
	if err != nil {
		eq.stats.FailedTotal++
	} else {
		eq.stats.ProcessedTotal++
	}

	eq.logger.Info("事件处理完成",
		"event_id", event.Event.EventID,
		"success", err == nil,
		"duration", record.Duration,
		"retry_count", event.Event.RetryCount)

	if err != nil {
		eq.logger.Error(err, "事件处理失败",
			"event_id", event.Event.EventID,
			"retry_count", event.Event.RetryCount)
	}
}

// 清理过期事件
func (eq *EventQueue) cleanupExpiredEvents(ctx context.Context) {
	ticker := time.NewTicker(5 * time.Minute)
	defer ticker.Stop()

	for {
		select {
		case <-ctx.Done():
			return
		case <-eq.stopCh:
			return
		case <-ticker.C:
			eq.cleanupExpired()
		}
	}
}

// 清理过期事件
func (eq *EventQueue) cleanupExpired() {
	eq.queueMutex.Lock()
	defer eq.queueMutex.Unlock()

	now := time.Now()
	cutoff := now.Add(-eq.config.EventTimeout)

	// 清理历史记录
	for eventID, record := range eq.history {
		if record.EndTime.Before(cutoff) {
			delete(eq.history, eventID)
		}
	}

	// 清理队列中的过期事件
	validEvents := make([]*QueuedEvent, 0, eq.queue.Len())
	for _, event := range *eq.queue {
		if event.EnqueuedAt.After(cutoff) {
			validEvents = append(validEvents, event)
		} else {
			eq.stats.ExpiredTotal++
		}
	}

	// 重建堆
	eq.queue = &EventHeap{}
	for _, event := range validEvents {
		heap.Push(eq.queue, event)
	}

	eq.stats.QueueSize = eq.queue.Len()
}

// 统计收集器
func (eq *EventQueue) statsCollector(ctx context.Context) {
	ticker := time.NewTicker(1 * time.Minute)
	defer ticker.Stop()

	for {
		select {
		case <-ctx.Done():
			return
		case <-eq.stopCh:
			return
		case <-ticker.C:
			eq.collectStats()
		}
	}
}

// 收集统计信息
func (eq *EventQueue) collectStats() {
	eq.queueMutex.RLock()
	defer eq.queueMutex.RUnlock()

	eq.stats.LastCollectTime = time.Now()
	eq.stats.QueueSize = eq.queue.Len()

	// 计算平均处理时间
	var totalDuration time.Duration
	var processedCount int

	for _, record := range eq.history {
		if record.Success {
			totalDuration += record.Duration
			processedCount++
		}
	}

	if processedCount > 0 {
		eq.stats.AverageProcessTime = totalDuration / time.Duration(processedCount)
	}

	eq.logger.Info("队列统计信息",
		"queue_size", eq.stats.QueueSize,
		"processed_total", eq.stats.ProcessedTotal,
		"failed_total", eq.stats.FailedTotal,
		"expired_total", eq.stats.ExpiredTotal,
		"avg_process_time", eq.stats.AverageProcessTime)
}

// 获取优先级数值
func (eq *EventQueue) getPriorityValue(priority Priority) int {
	switch priority {
	case PriorityCritical:
		return 4
	case PriorityHigh:
		return 3
	case PriorityMedium:
		return 2
	case PriorityLow:
		return 1
	default:
		return 0
	}
}
```

```go
	"time"

	"github.com/go-logr/logr"

// 集群状态机
type ClusterStateMachine struct {
	// 配置
	config *StateMachineConfig

	// 日志记录器

	// 当前状态
	currentState *ClusterState

	// 状态锁

	// 状态存储
	stateStore StateStore

	// 状态转换规则

	// 重试管理器
	retryManager *RetryManager

	// 事件通道

	// 停止通道

// 集群状态
type ClusterState struct {
	// 版本号
	Version int64 `json:"version"`

	// 时间戳
	Timestamp time.Time `json:"timestamp"`

	// 集群规格
	Spec ClusterSpec `json:"spec"`

	// 当前状态
	Status ClusterStatus `json:"status"`

	// 成员信息
	Members map[string]MemberInfo `json:"members"`

	// 当前操作
	CurrentOperation *OperationInfo `json:"current_operation,omitempty"`

	// 上次操作结果
	LastOperationResult *OperationResult `json:"last_operation_result,omitempty"`

	// 检查点信息
	Checkpoint *CheckpointInfo `json:"checkpoint,omitempty"`
}

// 状态类型
type StateType string

const (
	StateCreating    StateType = "creating"
	StateRunning     StateType = "running"
	StateUpdating    StateType = "updating"
	StateScaling     StateType = "scaling"
	StateRecovering  StateType = "recovering"
	StateFailed      StateType = "failed"
	StateTerminating StateType = "terminating"
)

// 事件类型
type EventType string

const (
	EventTypeStart           EventType = "start"
	EventTypeCreateComplete  EventType = "create_complete"
	EventTypeUpdateSpec      EventType = "update_spec"
	EventTypeScaleUp         EventType = "scale_up"
	EventTypeScaleDown       EventType = "scale_down"
	EventTypeMemberFailure   EventType = "member_failure"
	EventTypeRecoveryComplete EventType = "recovery_complete"
	EventTypeTerminate       EventType = "terminate"
	EventTypeError           EventType = "error"
)

// 状态事件
type StateEvent struct {
	// 事件类型
	Type EventType `json:"type"`

	// 事件数据
	Data interface{} `json:"data"`

	// 时间戳
	Timestamp time.Time `json:"timestamp"`

	// 源
	Source string `json:"source"`
}

// 操作信息
type OperationInfo struct {

	Type string `json:"type"`

	// 开始时间
	StartTime time.Time `json:"start_time"`

	// 进度
	Progress *OperationProgress `json:"progress"`

	// 重试次数
	RetryCount int `json:"retry_count"`
}

// 操作进度
type OperationProgress struct {

	Stage string `json:"stage"`

	// 完成百分比
	Percentage int `json:"percentage"`

	Message string `json:"message"`

// 新建状态机
func NewClusterStateMachine(config *StateMachineConfig, logger logr.Logger) *ClusterStateMachine {
	sm := &ClusterStateMachine{
		config:        config,
		logger:        logger,
		currentState:  &ClusterState{},
		stateStore:    config.StateStore,
		retryManager:  NewRetryManager(config.MaxRetries, config.BaseRetryDelay),
		eventCh:       make(chan StateEvent, 100),
		stopCh:        make(chan struct{}),
	}

	// 初始化状态转换规则
	sm.initTransitions()

	return sm
}

// 初始化状态转换规则
func (sm *ClusterStateMachine) initTransitions() {
	sm.transitions = map[StateType]map[EventType]StateType{
		StateCreating: {
			EventTypeCreateComplete: StateRunning,
			EventTypeError:         StateFailed,
		},
		StateRunning: {
			EventTypeUpdateSpec:    StateUpdating,
			EventTypeScaleUp:       StateScaling,
			EventTypeScaleDown:     StateScaling,
			EventTypeMemberFailure: StateRecovering,
			EventTypeTerminate:     StateTerminating,
			EventTypeError:         StateFailed,
		},
		StateUpdating: {
			EventTypeUpdateSpec:    StateUpdating,  // 自身转换
			EventTypeMemberFailure: StateRecovering,
			EventTypeError:         StateFailed,
		},
		StateScaling: {
			EventTypeScaleUp:       StateScaling,  // 自身转换
			EventTypeScaleDown:     StateScaling,  // 自身转换
			EventTypeMemberFailure: StateRecovering,
			EventTypeError:         StateFailed,
		},
		StateRecovering: {
			EventTypeRecoveryComplete: StateRunning,
			EventTypeMemberFailure:    StateRecovering,  // 自身转换
			EventTypeError:            StateFailed,
		},
		StateFailed: {
			EventTypeStart:         StateCreating,
			EventTypeTerminate:     StateTerminating,
		},
		StateTerminating: {
			EventTypeTerminate: StateTerminating,  // 自身转换
		},
	}
}

// 启动状态机
func (sm *ClusterStateMachine) Start(ctx context.Context) error {
	sm.logger.Info("启动集群状态机")

	// 尝试从存储中恢复状态
	if err := sm.recoverState(); err != nil {
		sm.logger.Error(err, "恢复状态失败，使用初始状态")
		sm.initializeInitialState()
	}

	// 启动事件处理器
	go sm.eventProcessor(ctx)

	// 启动定期检查点创建器
	go sm.checkpointCreator(ctx)

	return nil
}

// 事件处理器
func (sm *ClusterStateMachine) eventProcessor(ctx context.Context) {
	for {
		select {
		case <-ctx.Done():
			return
		case <-sm.stopCh:
			return
		case event := <-sm.eventCh:
			sm.processEvent(event)
		}
	}
}

// 处理事件
func (sm *ClusterStateMachine) processEvent(event StateEvent) {
	sm.stateLock.Lock()
	defer sm.stateLock.Unlock()

	currentStateType := sm.currentState.Status.Phase
	sm.logger.Info("处理状态事件",
		"current_state", currentStateType,
		"event_type", event.Type,
		"source", event.Source)

	// 检查状态转换是否合法
	targetState, isValid := sm.transitions[currentStateType][event.Type]
	if !isValid {
		sm.logger.Warn("无效的状态转换",
			"current_state", currentStateType,
			"event_type", event.Type)
		return
	}

	// 执行状态转换
	if err := sm.transitionToState(targetState, event); err != nil {
		sm.logger.Error(err, "状态转换失败",
			"from", currentStateType,
			"to", targetState)

		// 如果转换失败，尝试转换到失败状态
		sm.transitionToState(StateFailed, StateEvent{
			Type:      EventTypeError,
			Data:      err.Error(),
			Timestamp: time.Now(),
			Source:    "state_machine",
		})
		return
	}

	sm.logger.Info("状态转换成功",
		"from", currentStateType,
		"to", targetState)
}

// 转换到目标状态
func (sm *ClusterStateMachine) transitionToState(targetState StateType, event StateEvent) error {
	// 创建状态检查点
	if err := sm.createCheckpoint(); err != nil {
		sm.logger.Error(err, "创建检查点失败")
	}

	// 更新状态版本
	sm.currentState.Version++
	sm.currentState.Timestamp = time.Now()

	// 更新状态阶段
	sm.currentState.Status.Phase = targetState

	// 执行状态特定的处理
	switch targetState {
	case StateCreating:
		return sm.handleCreating(event)
	case StateRunning:
		return sm.handleRunning(event)
	case StateUpdating:
		return sm.handleUpdating(event)
	case StateScaling:
		return sm.handleScaling(event)
	case StateRecovering:
		return sm.handleRecovering(event)
	case StateFailed:
		return sm.handleFailed(event)
	case StateTerminating:
		return sm.handleTerminating(event)
	default:
		return fmt.Errorf("未知的状态类型: %s", targetState)
	}
}

// 处理创建状态
func (sm *ClusterStateMachine) handleCreating(event StateEvent) error {
	sm.logger.Info("处理创建状态")

	// 设置当前操作
	sm.currentState.CurrentOperation = &OperationInfo{
		Type:       "create_cluster",
		StartTime:  time.Now(),
		RetryCount: 0,
		Progress: &OperationProgress{
			Stage:       "initializing",
			Percentage:  0,
			Message:     "开始创建集群",
			LastUpdated: time.Now(),
		},
	}

	// 更新进度
	sm.updateOperationProgress("creating_resources", 20)

	// 持久化状态
	if err := sm.persistState(); err != nil {
		return fmt.Errorf("持久化状态失败: %w", err)
	}

	return nil
}

// 处理运行状态
func (sm *ClusterStateMachine) handleRunning(event StateEvent) error {
	sm.logger.Info("处理运行状态")

	// 清除当前操作
	sm.currentState.CurrentOperation = nil

	// 设置上次操作结果
	sm.currentState.LastOperationResult = &OperationResult{
		Success:   true,
		Timestamp: time.Now(),
		Message:   "集群运行正常",
	}

	// 持久化状态
	if err := sm.persistState(); err != nil {
		return fmt.Errorf("持久化状态失败: %w", err)
	}

	return nil
}

// 处理扩缩容状态
func (sm *ClusterStateMachine) handleScaling(event StateEvent) error {
	sm.logger.Info("处理扩缩容状态")

	data, ok := event.Data.(*ScalingData)
	if !ok {
		return fmt.Errorf("无效的扩缩容数据")
	}

	// 设置当前操作
	sm.currentState.CurrentOperation = &OperationInfo{
		Type:       "scaling",
		StartTime:  time.Now(),
		RetryCount: 0,
		Progress: &OperationProgress{
			Stage:       "scaling_cluster",
			Percentage:  0,
			Message:     fmt.Sprintf("扩缩容到 %d 个节点", data.TargetSize),
			LastUpdated: time.Now(),
		},
	}

	// 更新集群规格
	sm.currentState.Spec.Size = data.TargetSize

	// 更新进度
	sm.updateOperationProgress("updating_members", 50)

	// 持久化状态
	if err := sm.persistState(); err != nil {
		return fmt.Errorf("持久化状态失败: %w", err)
	}

	return nil
}

// 处理故障恢复状态
func (sm *ClusterStateMachine) handleRecovering(event StateEvent) error {
	sm.logger.Info("处理故障恢复状态")

	data, ok := event.Data.(*RecoveryData)
	if !ok {
		return fmt.Errorf("无效的故障恢复数据")

	// 设置当前操作
	sm.currentState.CurrentOperation = &OperationInfo{

		StartTime:  time.Now(),

		Progress: &OperationProgress{
			Stage:       "recovering_members",

			Message:     fmt.Sprintf("恢复 %d 个故障节点", len(data.FailedMembers)),
			LastUpdated: time.Now(),
		},
	}

	// 更新进度
	sm.updateOperationProgress("isolating_failed_members", 30)

	// 持久化状态
	if err := sm.persistState(); err != nil {
		return fmt.Errorf("持久化状态失败: %w", err)
	}

	return nil
}

// 处理失败状态
func (sm *ClusterStateMachine) handleFailed(event StateEvent) error {
	sm.logger.Info("处理失败状态")

	// 设置错误信息
	errorMsg, ok := event.Data.(string)
	if !ok {
		errorMsg = "未知错误"
	}

	// 设置当前操作
	sm.currentState.CurrentOperation = &OperationInfo{
		Type:       "error",
		StartTime:  time.Now(),
		RetryCount: 0,
		Progress: &OperationProgress{
			Stage:       "error",
			Percentage:  100,
			Message:     errorMsg,
			LastUpdated: time.Now(),
		},
	}

	// 设置上次操作结果
	sm.currentState.LastOperationResult = &OperationResult{
		Success:   false,
		Timestamp: time.Now(),
		Message:   errorMsg,
	}

	// 持久化状态
	if err := sm.persistState(); err != nil {
		return fmt.Errorf("持久化状态失败: %w", err)
	}

	return nil
}

// 发送事件
func (sm *ClusterStateMachine) SendEvent(event StateEvent) {
	select {
	case sm.eventCh <- event:
		sm.logger.Info("事件已发送到状态机",
			"event_type", event.Type,
			"source", event.Source)
	default:
		sm.logger.Warn("事件队列已满，丢弃事件",
			"event_type", event.Type,
			"source", event.Source)
	}
}

// 更新操作进度
func (sm *ClusterStateMachine) updateOperationProgress(stage string, percentage int) {
	if sm.currentState.CurrentOperation != nil {
		sm.currentState.CurrentOperation.Progress.Stage = stage
		sm.currentState.CurrentOperation.Progress.Percentage = percentage
		sm.currentState.CurrentOperation.Progress.LastUpdated = time.Now()
	}
}

// 创建检查点
func (sm *ClusterStateMachine) createCheckpoint() error {
	sm.currentState.Checkpoint = &CheckpointInfo{
		Version:   sm.currentState.Version,
		Timestamp: time.Now(),
		State:     sm.currentState.Status.Phase,
		Operation: sm.currentState.CurrentOperation,
	}

	if err := sm.stateStore.SaveCheckpoint(sm.currentState.Checkpoint); err != nil {
		return fmt.Errorf("保存检查点失败: %w", err)
	}

	return nil
}

// 持久化状态
func (sm *ClusterStateMachine) persistState() error {
	return sm.stateStore.SaveState(sm.currentState)
}

// 恢复状态
func (sm *ClusterStateMachine) recoverState() error {
	state, err := sm.stateStore.LoadState()
	if err != nil {
		return fmt.Errorf("加载状态失败: %w", err)
	}

	if state != nil {
		sm.currentState = state
		sm.logger.Info("状态恢复成功",
			"version", state.Version,
			"state", state.Status.Phase)
	}

	return nil
}

// 初始化初始状态
func (sm *ClusterStateMachine) initializeInitialState() {
	sm.currentState = &ClusterState{
		Version: 1,
		Spec: ClusterSpec{
			Size:     sm.config.InitialSize,
			Version:  sm.config.InitialVersion,
			Replicas: sm.config.InitialSize,
		},
		Status: ClusterStatus{
			Phase:   StateCreating,
			Members: make(map[string]MemberInfo),
		},
		Members: make(map[string]MemberInfo),
		Timestamp: time.Now(),
	}
}

// 定期检查点创建器
func (sm *ClusterStateMachine) checkpointCreator(ctx context.Context) {
	ticker := time.NewTicker(sm.config.CheckpointInterval)
	defer ticker.Stop()

	for {
		select {
		case <-ctx.Done():
			return
		case <-sm.stopCh:
			return
		case <-ticker.C:
			if err := sm.createCheckpoint(); err != nil {
				sm.logger.Error(err, "创建定期检查点失败")
			}
		}
	}
}

// 获取当前状态
func (sm *ClusterStateMachine) GetCurrentState() *ClusterState {
	sm.stateLock.RLock()
	defer sm.stateLock.RUnlock()

	// 返回状态的副本
	stateCopy := *sm.currentState
	return &stateCopy
}

// 停止状态机
func (sm *ClusterStateMachine) Stop() {
	close(sm.stopCh)
}
```
